# The Trash place lists what went to the trash. Its row menu restores (to where the item was), deletes for
# good and empties the trash; each delete asks first.
click place Trash
expect folder trash:///
expect rows 0
key backspace
expect folder .

# three items to the trash
key G
expect cursor zeta.md
key D
key enter
key g
key g
expect cursor alpha
key D
key enter
key G
expect cursor report.tar.gz
key D
key enter
expect missing zeta.md
expect missing alpha
expect missing report.tar.gz

click place Trash
expect folder trash:///
expect rows 3
shot trash.png

# Restore, the menu's first item
rclick row zeta.md
key down
key enter
expect exists zeta.md
expect missing ../data/Trash/files/zeta.md
expect missing ../data/Trash/info/zeta.md.trashinfo
expect rows 2
expect toast Restored to

# a folder comes back whole
rclick row alpha
key down
key enter
expect exists alpha/sub/deep.txt
expect rows 1
