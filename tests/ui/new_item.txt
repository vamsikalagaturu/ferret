# a: a new row at the top to name a file, or a folder when the name ends in /; Enter makes it, Esc drops it.
key a
expect mode INSERT
type todo.txt
key enter
expect mode NORMAL
expect exists todo.txt
expect cursor todo.txt

key a
type stuff/
key enter
expect exists stuff
expect cursor stuff
key l
expect folder stuff
key h

key a
type never.txt
key esc
expect mode NORMAL
expect missing never.txt
expect cursor alpha
key j
expect cursor beta
