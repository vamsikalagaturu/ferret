# Files compiled in as byte arrays: <symbol>[] and <symbol>_size (NUL-terminated).
function(fr_embed out symbol file)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
  file(READ ${file} hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
  file(WRITE ${out}.tmp "const unsigned char ${symbol}[] = {${bytes}0x00};\nconst int ${symbol}_size = sizeof(${symbol}) - 1;\n")
  configure_file(${out}.tmp ${out} COPYONLY)
endfunction()

# The fonts and the Lucide icons, as FR_ASSET_SOURCES with a table of icons by name (src/ui/icons.h).
function(fr_embed_assets gen)
  fr_embed(${gen}/font_regular.c fr_font_regular ${FR_ASSETS}/Inter-Regular.ttf)
  fr_embed(${gen}/font_semibold.c fr_font_semibold ${FR_ASSETS}/Inter-SemiBold.ttf)
  set(sources ${gen}/font_regular.c ${gen}/font_semibold.c)
  file(GLOB icons ${FR_ASSETS}/icons/*.svg)
  set(table "")
  set(decls "")
  foreach(icon ${icons})
    get_filename_component(name ${icon} NAME_WE)
    string(MAKE_C_IDENTIFIER ${name} id)
    fr_embed(${gen}/icon_${id}.c fr_icon_${id} ${icon})
    list(APPEND sources ${gen}/icon_${id}.c)
    string(APPEND decls "extern const unsigned char fr_icon_${id}[];\nextern const int fr_icon_${id}_size;\n")
    string(APPEND table "  {\"${name}\", fr_icon_${id}, &fr_icon_${id}_size},\n")
  endforeach()
  file(WRITE ${gen}/icons_table.c.tmp "#include \"icons.h\"\n${decls}const fr_icon_data fr_icon_files[] = {\n${table}  {0, 0, 0}\n};\n")
  configure_file(${gen}/icons_table.c.tmp ${gen}/icons_table.c COPYONLY)
  list(APPEND sources ${gen}/icons_table.c)
  set(FR_ASSET_SOURCES ${sources} PARENT_SCOPE)
endfunction()

# libarchive, static, from platform/*/deps.sh; a codec built beside it is linked from there, else the system's
set(FR_LIBARCHIVE_DIR "" CACHE PATH "libarchive's install prefix (include/archive.h, lib/libarchive.a)")
if(NOT EXISTS "${FR_LIBARCHIVE_DIR}/lib/libarchive.a")
  message(FATAL_ERROR "FR_LIBARCHIVE_DIR: no lib/libarchive.a in '${FR_LIBARCHIVE_DIR}' (platform/*/deps.sh builds it)")
endif()
set(FR_ARCHIVE_LIBS ${FR_LIBARCHIVE_DIR}/lib/libarchive.a)
foreach(codec zstd lzma bz2 z)
  if(EXISTS ${FR_LIBARCHIVE_DIR}/lib/lib${codec}.a)
    list(APPEND FR_ARCHIVE_LIBS ${FR_LIBARCHIVE_DIR}/lib/lib${codec}.a)
  else()
    list(APPEND FR_ARCHIVE_LIBS ${codec})
  endif()
endforeach()
target_include_directories(fercore SYSTEM PUBLIC ${FR_LIBARCHIVE_DIR}/include)
target_compile_definitions(fercore PUBLIC LIBARCHIVE_STATIC)
target_link_libraries(fercore PUBLIC ${FR_ARCHIVE_LIBS})

# SMB and SFTP (src/net, src/vfs/vfs_{smb,sftp}.c): libsmb2 and libssh2 as platform/*/deps.sh builds them, over
# OpenSSL's libcrypto, built there too (Android) or the system's
set(FR_NET_PREFIX "" CACHE PATH "Where platform/*/deps.sh installed libsmb2 (smb2/), libssh2 (ssh2/) and OpenSSL (openssl/)")
if(NOT EXISTS "${FR_NET_PREFIX}/smb2/lib/libsmb2.a" OR NOT EXISTS "${FR_NET_PREFIX}/ssh2/lib/libssh2.a")
  message(FATAL_ERROR "FR_NET_PREFIX ('${FR_NET_PREFIX}') has no libsmb2 and libssh2: platform/*/build.sh builds them")
endif()
foreach(lib smb2 ssh2)
  add_library(fr_${lib} STATIC IMPORTED GLOBAL)
  set_target_properties(fr_${lib} PROPERTIES IMPORTED_LOCATION ${FR_NET_PREFIX}/${lib}/lib/lib${lib}.a
                        INTERFACE_INCLUDE_DIRECTORIES ${FR_NET_PREFIX}/${lib}/include)
endforeach()
if(EXISTS "${FR_NET_PREFIX}/openssl/lib/libcrypto.a")
  add_library(fr_crypto STATIC IMPORTED GLOBAL)
  set_target_properties(fr_crypto PROPERTIES IMPORTED_LOCATION ${FR_NET_PREFIX}/openssl/lib/libcrypto.a
                        INTERFACE_INCLUDE_DIRECTORIES ${FR_NET_PREFIX}/openssl/include)
  set_property(TARGET fr_ssh2 APPEND PROPERTY INTERFACE_LINK_LIBRARIES fr_crypto)
  add_library(fr_ssl STATIC IMPORTED GLOBAL)
  set_target_properties(fr_ssl PROPERTIES IMPORTED_LOCATION ${FR_NET_PREFIX}/openssl/lib/libssl.a
                        INTERFACE_INCLUDE_DIRECTORIES ${FR_NET_PREFIX}/openssl/include INTERFACE_LINK_LIBRARIES fr_crypto)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBCRYPTO REQUIRED IMPORTED_TARGET GLOBAL libcrypto)
  set_property(TARGET fr_ssh2 APPEND PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::LIBCRYPTO)
  pkg_check_modules(LIBSSL REQUIRED IMPORTED_TARGET GLOBAL libssl)
  add_library(fr_ssl INTERFACE IMPORTED GLOBAL)
  set_property(TARGET fr_ssl PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::LIBSSL)
endif()
target_link_libraries(fercore PUBLIC fr_smb2 fr_ssh2)

# FTP/FTPS, WebDAV and NFS (src/vfs/vfs_{curl,ftp,dav,nfs}.c): libcurl, expat and libnfs from FR_NET_PREFIX's curl/,
# expat/ and nfs/ where deps.sh built them, else the system's. ferret checks TLS certificates through libcurl's
# OpenSSL (its SSL_CTX), so libssl is linked directly too.
function(fr_net_lib name pkg)
  if(EXISTS "${FR_NET_PREFIX}/${name}/lib/lib${name}.a")
    add_library(fr_${name} STATIC IMPORTED GLOBAL)
    set_target_properties(fr_${name} PROPERTIES IMPORTED_LOCATION ${FR_NET_PREFIX}/${name}/lib/lib${name}.a
                          INTERFACE_INCLUDE_DIRECTORIES ${FR_NET_PREFIX}/${name}/include)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FR_PC_${name} REQUIRED IMPORTED_TARGET GLOBAL ${pkg})
    add_library(fr_${name} INTERFACE IMPORTED GLOBAL)
    set_property(TARGET fr_${name} PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::FR_PC_${name})
  endif()
endfunction()
fr_net_lib(curl libcurl)
fr_net_lib(expat expat)
fr_net_lib(nfs libnfs)
if(EXISTS "${FR_NET_PREFIX}/curl/lib/libcurl.a")
  set_property(TARGET fr_curl APPEND PROPERTY INTERFACE_LINK_LIBRARIES fr_ssl)
  set_property(TARGET fr_curl APPEND PROPERTY INTERFACE_COMPILE_DEFINITIONS CURL_STATICLIB)
endif()
target_link_libraries(fercore PUBLIC fr_curl fr_expat fr_nfs fr_ssl)

# A release (FR_RELEASE=1 in the environment) is the plain version; any other build is a development one, as semver
# pre-release + build metadata: 0.1.0-dev+1b42e42, .dirty with uncommitted changes. Fixed when configured.
set(FR_VERSION_STRING "${PROJECT_VERSION}")
if(NOT "$ENV{FR_RELEASE}" STREQUAL "1")
  set(FR_VERSION_STRING "${PROJECT_VERSION}-dev")
  find_package(Git QUIET)
  if(GIT_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --short=7 HEAD WORKING_DIRECTORY "${FR_REPO}"
                    OUTPUT_VARIABLE FR_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE FR_GIT_FAILED)
    execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain --untracked-files=no
                    WORKING_DIRECTORY "${FR_REPO}" OUTPUT_VARIABLE FR_GIT_CHANGES ERROR_QUIET)
    if(NOT FR_GIT_FAILED AND FR_GIT_COMMIT)
      set(FR_VERSION_STRING "${FR_VERSION_STRING}+${FR_GIT_COMMIT}")
      if(FR_GIT_CHANGES)
        set(FR_VERSION_STRING "${FR_VERSION_STRING}.dirty")
      endif()
    endif()
  endif()
endif()
message(STATUS "ferret ${FR_VERSION_STRING}")
target_compile_definitions(fercore PUBLIC FR_VERSION="${FR_VERSION_STRING}")
